\Q\mk{A}
\begin{parts}
\item Find the gradient of $f(x)=x^{4}-3x^{2}+2x$ at the point where $x=-1$.
\ALine{$f'(x)=4x^{3}-6x+2$;\ \ $f'(-1)=-4+6+2=4$}
\item Find the equation of the \ttangent to $y=2x^{2}-5x+1$ at $x=3$.
\ALineT{Point: $y=18-15+1=4$;\ \ $\dfrac{\mathrm{d}y}{\mathrm{d}x}=4x-5$, so $m=7$}
\ALine{$y-4=7(x-3)$, so $y=7x-17$}
\item $f'(x)=3x^{2}-8x+2$ and $f(2)=1$. Find $f(x)$.
\ALine{$f(x)=x^{3}-4x^{2}+2x+c$;\ \ $8-16+4+c=1$, so $c=5$}
\ALine{$f(x)=x^{3}-4x^{2}+2x+5$}
\item For which values of $x$ is $f(x)=x^{3}-6x^{2}+5$ decreasing?
\ALine{$f'(x)=3x^{2}-12x=3x(x-4)<0$}
\ALine{Negative between the roots: $0<x<4$}
\end{parts}

\Q\mk{M}
\begin{parts}
\item A particle moves with $s=4t^{3}-15t^{2}+12t$ metres after $t$ seconds. Find the times when it is at rest.
\ALine{$v=12t^{2}-30t+12=6(2t^{2}-5t+2)=6(2t-1)(t-2)$}
\ALine{At rest at $t=0.5$ s and $t=2$ s}
\item A drone takes off from the ground with vertical velocity $v=6t-t^{2}\mps$ for $0\le t\le6$. Find its maximum height, and justify that it is the maximum.
\ALineT{$s=3t^{2}-\dfrac{1}{3}t^{3}$ (from the ground, $c=0$)}
\ALine{$v=t(6-t)\ge0$ for $0\le t\le6$, so the drone rises the whole time;}
\ALine{$v=0$ at $t=6$: maximum height $s(6)=108-72=36$ m}
\end{parts}
