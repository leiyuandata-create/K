\sheethead{Lesson 1 \textperiodcentered\ Homework}{NCEA Level 2 \textperiodcentered\ AS 91262 \textperiodcentered\ Gradients, differentiation and tangents}{about 60 minutes}{}
Show all working and write $f'(x)$ each time. [A] Achieved, [M] Merit, [E] Excellence.

\Q Differentiate.\mk{A}
\begin{parts}
\item $f(x)=4x^{5}-2x^{3}+9$
\ALine{$f'(x)=20x^{4}-6x^{2}$}
\item $y=7-3x+5x^{2}$
\ALineT{$\dydx=-3+10x$}
\item $g(x)=\dfrac{1}{2}x^{4}-\dfrac{1}{3}x^{3}+x$
\ALine{$g'(x)=2x^{3}-x^{2}+1$}
\item $y=(x+4)(x-1)$
\ALine{$y=x^{2}+3x-4$}
\ALineT{$\dydx=2x+3$}
\end{parts}

\Q Differentiate $y=x^{2}(3x-5)$.\mk{A}
\ALine{$y=3x^{3}-5x^{2}$}
\ALineT{$\dydx=9x^{2}-10x$}

\Q Differentiate $f(x)=(2x-3)^{2}$.\mk{A}
\ALine{$f(x)=4x^{2}-12x+9$}
\ALine{$f'(x)=8x-12$}

\Q Find the gradient of $f(x)=x^{3}+2x^{2}-5x+1$ at the point where $x=2$.\mk{A}
\ALine{$f'(x)=3x^{2}+4x-5$}
\ALine{$f'(2)=12+8-5=15$}

\Q Find the gradient of $y=4-3x^{2}$ at the point where $x=-2$.\mk{A}
\ALineT{$\dydx=-6x$;\ at $x=-2$:\ $-6(-2)=12$}
\ALine{}

\Q For $y=x^{4}-8x$, find the gradient at $x=1$. Is the curve going up or down there?\mk{A}
\ALineT{$\dydx=4x^{3}-8$;\ at $x=1$:\ $4-8=-4$}
\ALine{Negative gradient, so the curve is going down at $x=1$.}

\Q Find the coordinates of the point on $y=x^{2}-6x+10$ where the gradient is $4$.\mk{A}
\ALine{$2x-6=4$, so $x=5$}
\ALine{$y=25-30+10=5$;\ the point is $(5,\,5)$}

\Q Find the $x$-values where $f(x)=x^{3}-3x^{2}-9x+2$ has gradient $0$.\mk{A}
\ALine{$f'(x)=3x^{2}-6x-9=0$}
\ALine{$x^{2}-2x-3=0$, so $(x-3)(x+1)=0$}
\ALine{$x=3$ or $x=-1$}

\Q Find the coordinates of the two points on $y=x^{3}-12x+1$ where the gradient is $15$.\mk{A}
\ALine{$3x^{2}-12=15$, so $x^{2}=9$, $x=\pm3$}
\ALine{$x=3$:\ $y=27-36+1=-8$;\quad $x=-3$:\ $y=-27+36+1=10$}
\ALine{Points $(3,\,-8)$ and $(-3,\,10)$}

\Q The curve $f(x)=ax^{3}+2x^{2}-1$ has gradient $32$ at $x=2$. Find $a$.\mk{A}
\ALine{$f'(x)=3ax^{2}+4x$}
\ALine{$f'(2)=12a+8=32$, so $a=2$}

\Q Find the equation of the tangent to $y=3x^{2}-x+2$ at $x=1$.\mk{A}
\ALine{Point: $y=3-1+2=4$, so $(1,\,4)$}
\ALineT{$\dydx=6x-1$, so $m=5$}
\ALine{$y-4=5(x-1)$, so $y=5x-1$}

\Q Find the equation of the tangent to $f(x)=x^{3}-4x$ at $x=-1$.\mk{A}
\ALine{Point: $f(-1)=-1+4=3$, so $(-1,\,3)$}
\ALine{$f'(x)=3x^{2}-4$, so $m=3-4=-1$}
\ALine{$y-3=-1(x+1)$, so $y=-x+2$}

\Q The curve $y=px^{2}+qx$ passes through $(2,\,2)$, and its gradient at that point is $5$. Find $p$ and $q$.\mk{M}
\ALine{Point: $4p+2q=2$}
\ALineT{Gradient: $\dydx=2px+q$, so $4p+q=5$}
\ALine{Subtract: $q=-3$;\ then $4p=8$, $p=2$}

\Q The tangent to $y=x^{2}+2x-3$ at the point P is parallel to the line $y=6x-1$. Find the coordinates of P and the equation of the tangent.\mk{M}
\ALine{Parallel means gradient $6$:\ $2x+2=6$, so $x=2$}
\ALine{$y=4+4-3=5$, so P is $(2,\,5)$}
\ALine{$y-5=6(x-2)$, so $y=6x-7$}

\QX The tangent to $y=x^{2}$ at the point where $x=a$ passes through $(0,\,-9)$. Find the possible values of $a$ and the equations of these tangents.\mk{E}
\ALine{Point $(a,\,a^{2})$; gradient $2a$; tangent: $y-a^{2}=2a(x-a)$, so $y=2ax-a^{2}$}
\ALine{Through $(0,\,-9)$:\ $-9=0-a^{2}$, so $a^{2}=9$, $a=3$ or $a=-3$}
\ALine{$a=3$:\ $y=6x-9$;\quad $a=-3$:\ $y=-6x-9$}
\ALine{}
