\Q Differentiate $f(x)=6x^{4}-3x^{3}+2x-11$.\mk{A}
\ALine{$f'(x)=24x^{3}-9x^{2}+2$}
\ALine{}

\Q Find the gradient of $y=x^{3}-4x^{2}+5$ at the point where $x=-1$.\mk{A}
\ALineT{$\dydx=3x^{2}-8x$}
\ALineT{At $x=-1$:\ $3(-1)^{2}-8(-1)=3+8=11$, so the gradient is $11$}
\ALine{}

\Q Find the coordinates of the point on $f(x)=3x^{2}-12x+2$ where the gradient is $6$.\mk{A}
\ALine{$f'(x)=6x-12=6$, so $x=3$}
\ALine{$f(3)=27-36+2=-7$}
\ALine{The point is $(3,\,-7)$}

\Q Find the equation of the tangent to $y=x^{3}-3x^{2}+2$ at $x=3$.\mk{A}
\ALine{Point: $y=27-27+2=2$, so $(3,\,2)$}
\ALineT{Gradient: $\dfrac{\mathrm{d}y}{\mathrm{d}x}=3x^{2}-6x$, so $m=27-18=9$}
\ALine{$y-2=9(x-3)$}
\ALine{$y=9x-25$}
