\Q $f'(x)=12x^{2}-2x+3$ and $f(1)=10$. Find $f(x)$.\mk{A}
\ALine{$f(x)=4x^{3}-x^{2}+3x+c$}
\ALine{$f(1)=4-1+3+c=10$, so $c=4$}
\ALine{$f(x)=4x^{3}-x^{2}+3x+4$}

\Q A particle has velocity $v=8-2t\mps$ after $t$ seconds, and starts $5$ m from the origin. Find its displacement at $t=3$, and when it is at rest.\mk{A}
\ALine{$s=8t-t^{2}+c$;\ \ $s(0)=5$, so $s=8t-t^{2}+5$}
\ALine{$s(3)=24-9+5=20$ m}
\ALine{At rest: $8-2t=0$, so $t=4$ s}
