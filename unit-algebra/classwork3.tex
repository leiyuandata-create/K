\section*{Part A \textperiodcentered{} Solve, count, build}
\qpart{A1}Solve $(2x - 1)(x + 4) = 0$.\lvl{A}
\af{x = \frac{1}{2}\quad\text{or}\quad x = -4}

\qpart{A2}Solve $x^{2} - 3x = 28$.\lvl{A}
\al{$x^{2} - 3x - 28 = 0$,\quad $(x - 7)(x + 4) = 0$,\quad $x = 7$ or $x = -4$}

\qpart{A3}Solve $3x^{2} + 2x - 4 = 0$.\lvl{A}
\af{x = \frac{-2 \pm \sqrt{(2)^{2} - 4(3)(-4)}}{2(3)} = \frac{-2 \pm \sqrt{52}}{6}}
\al{$x = 0.869$ or $x = -1.535$}

\qpart{A4}Find the discriminant of $x^{2} - 4x + 7 = 0$ and say what it tells you.\lvl{A}
\al{$(-4)^{2} - 4(1)(7) = -12 < 0$, so there are no real roots}

\qpart{A5}Write a quadratic equation with roots $\dfrac{2}{3}$ and $-1$, with whole-number coefficients.\lvl{A}
\al{$(3x - 2)(x + 1) = 0$,\quad $3x^{2} + x - 2 = 0$}

\section*{Part B \textperiodcentered{} Find the unknown}
\qpart{B1}$3x^{2} + kx + 12 = 0$ has equal roots. Find $k$.\lvl{M}
\al{$k^{2} - 4(3)(12) = 0$,\quad $k^{2} = 144$,\quad $k = 12$ or $k = -12$}

\qpart{B2}$x^{2} + 8x + p = 0$ has two different real roots. Find the values of $p$.\lvl{M}
\al{$64 - 4p > 0$,\quad $-4p > -64$,\quad $p < 16$}

\qpart{B3}A parabola crosses the $x$-axis at $-3$ and $2$ and passes through $(1, -8)$. Find its equation.\lvl{M}
\al{$y = a(x + 3)(x - 2)$;\quad at $(1, -8)$:\quad $-8 = a(4)(-1)$,\quad $a = 2$}
\al{$y = 2(x + 3)(x - 2) = 2x^{2} + 2x - 12$}

\section*{Part C \textperiodcentered{} Real roots for every $k$}
\qpart{C}Show that $x^{2} - (k + 3)x + 3k = 0$ has real roots for every value of $k$, and find when the roots are equal.\lvl{E}
\begin{plainbox}[title={Words you may use}]
discriminant \textperiodcentered{} perfect square \textperiodcentered{} never negative \textperiodcentered{} therefore \textperiodcentered{} equal roots
\end{plainbox}
Sentence starters (optional): \emph{The discriminant is \ldots} \quad \emph{This is a perfect square, so \ldots} \quad \emph{The roots are equal when \ldots}
\al{$\Delta = \bigl(-(k + 3)\bigr)^{2} - 4(1)(3k) = k^{2} + 6k + 9 - 12k$ \hfill MP1}
\al{$= k^{2} - 6k + 9 = (k - 3)^{2}$ \hfill MP2}
\al{A square is never negative, so $\Delta \geq 0$: the roots are real for every $k$. \hfill MP3}
\al{The roots are equal when $\Delta = 0$, that is when $k = 3$ (then $(x - 3)^{2} = 0$).}
\ansnote{6mm}{\small MP1 is r; MP2 with the reasoning of MP3 is t.}

\section*{Extension}
Solve $x^{4} - 5x^{2} + 4 = 0$.
\al{Let $u = x^{2}$:\quad $(u - 1)(u - 4) = 0$,\quad $x^{2} = 1$ or $4$,\quad $x = \pm 1$ or $x = \pm 2$}
\al{}
