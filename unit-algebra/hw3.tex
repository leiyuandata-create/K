\QUESTION{TWO}
\qpart{(a)}Write an equation $x^{2} + bx + c = 0$ with roots $-3$ and $5$.\lvl{A}
\al{$(x + 3)(x - 5) = 0$,\quad $x^{2} - 2x - 15 = 0$}

\qpart{(b)}Write an equation with roots $\dfrac{2}{5}$ and $-3$, with whole-number coefficients.\lvl{A}
\al{$(5x - 2)(x + 3) = 0$,\quad $5x^{2} + 13x - 6 = 0$}

\qpart{(c)}A parabola crosses the $x$-axis at $-4$ and $1$ and passes through $(0, 8)$. Find its equation.\lvl{M}
\al{$y = a(x + 4)(x - 1)$;\quad $8 = a(4)(-1)$,\quad $a = -2$}
\al{$y = -2(x + 4)(x - 1) = -2x^{2} - 6x + 8$}

\qpart{(d)}Show that the roots of $3x^{2} - 10kx + 3k^{2} = 0$ ($k \neq 0$) are $3k$ and $\dfrac{k}{3}$.\lvl{M}
\alx{$(3x - k)(x - 3k) = 0$,\quad so $x = \dfrac{k}{3}$ or $x = 3k$}

\qpart{(e)}$x^{2} + bx + 18 = 0$ has one root twice the other. Find the possible values of $b$.\lvl{E}
\al{Let the roots be $r$ and $2r$:\quad $(x - r)(x - 2r) = x^{2} - 3rx + 2r^{2}$}
\al{So $2r^{2} = 18$, $r = 3$ or $r = -3$, and $b = -3r$:\quad $b = -9$ or $b = 9$}
\al{Check $b = -9$: $x^{2} - 9x + 18 = (x - 3)(x - 6)$, roots 3 and 6.}
\gradescore{u: (a), (b) correct; (c) set up. r: (c) correct; (d) shown with the factorisation.\\
t: (e) both values of $b$ with reasoning.\quad N1 partial; N2 1u; A3 2u; A4 3u; M5 1r; M6 2r; E7 t, minor error; E8 t.}

\QUESTION{THREE}
\qpart{(a)}Find the discriminant of $3x^{2} - 5x + 4 = 0$ and state the number of real roots.\lvl{A}
\al{$(-5)^{2} - 4(3)(4) = 25 - 48 = -23 < 0$:\quad no real roots}

\qpart{(b)}Find $k$ so that $x^{2} - 10x + k = 0$ has equal roots.\lvl{A}
\al{$(-10)^{2} - 4(1)(k) = 0$,\quad $100 = 4k$,\quad $k = 25$}

\qpart{(c)}Find the values of $m$ for which $2x^{2} + mx + 18 = 0$ has no real roots.\lvl{M}
\al{$m^{2} - 4(2)(18) < 0$,\quad $m^{2} < 144$,\quad $-12 < m < 12$}

\qpart{(d)}$kx^{2} + 6x + (k + 8) = 0$ has equal roots. Find $k$.\lvl{M}
\al{$36 - 4k(k + 8) = 0$,\quad $k^{2} + 8k - 9 = 0$,\quad $(k + 9)(k - 1) = 0$,\quad $k = 1$ or $k = -9$}

\qpart{(e)}Show that $x^{2} + (2k - 1)x + k^{2} - k = 0$ has two different real roots for every value of $k$.\lvl{E}
\al{$\Delta = (2k - 1)^{2} - 4(1)(k^{2} - k) = 4k^{2} - 4k + 1 - 4k^{2} + 4k = 1$}
\al{$\Delta = 1 > 0$ whatever $k$ is, so there are always two different real roots.}
\gradescore{u: (a), (b) correct. r: (c) range found; (d) both values.\\
t: (e) $\Delta = 1$ with the conclusion for every $k$.\quad N1 partial; N2 1u; A3 2u; A4 3u; M5 1r; M6 2r; E7 t, minor error; E8 t.}
